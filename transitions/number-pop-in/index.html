Number pop-in

Namespaced classes and motion tokens you can retune. The reduced-motion guard ships with the snippet.

/* Transitions.dev — Number pop-in */

/*
  Usage:
    <span class="t-digit-group is-animating">
      <span class="t-digit" data-same>1</span>
      <span class="t-digit-slot">
        <span class="t-digit-old" aria-hidden="true">2</span>
        <span class="t-digit">3</span>
      </span>
      <span class="t-digit-slot" data-stagger="1">
        <span class="t-digit-old" aria-hidden="true">4</span>
        <span class="t-digit">5</span>
      </span>
    </span>

  Each changed digit gets a .t-digit-slot holding the old digit
  (leaves upward) and the new one (rises just after it, pulls back,
  springs into place past 100%). --digit-dur covers anticipation and
  spring.

  Replay:
    - Remove `.is-animating`, re-render the digits, force a reflow,
      then re-add `.is-animating`.
    - Use data-stagger="1", "2" on a slot to delay it by
      `n * var(--digit-stagger)`.
    - Mark the leading digits the old and new number share with
      data-same: they stay still and only the changed digits swap
      (1,299 to 1,305: "1," stays, "305" swaps).

  Sizes: distances are in em and --digit-blur is a curve on the font
  size, so the motion fits any number size with no changes.
*/

  --digit-enter-delay: 50ms;

// Pair with the CSS from the CSS tab. Changed digits get a slot with the old
// digit (leaves upward) and the new one (rises just after it, springs into
// place). The leading characters both numbers share stay still.
export function NumberPopIn({ initial = "65.78" }) {
  const [value, setValue] = useState(initial);
  const prev = useRef(value);
  const from = prev.current;
  useEffect(() => { prev.current = value; }, [value]);

  let same = 0;
  while (same < from.length && same < value.length && from[same] === value[same]) same++;

  const next = () => {
    const whole = Math.floor(Math.random() * 90) + 10;
    const part = String(Math.floor(Math.random() * 100)).padStart(2, "0");
    setValue(whole + "." + part);
  };

  return (
    <>
      <button type="button" onClick={next}>Animate</button>
      {/* The key restarts the animation on every change. */}
      <span key={value} className={"t-digit-group" + (from !== value ? " is-animating" : "")}>
        {value.split("").map((ch, i) => {
          if (i < same || !/\d/.test(ch)) {
            return <span key={i} className="t-digit" data-same="">{ch}</span>;
          }
          const fromEnd = value.length - 1 - i;
          const old = from[from.length - (value.length - i)];
          return (
            <span
              key={i}
              className="t-digit-slot"
              data-stagger={fromEnd === 1 ? "1" : fromEnd === 0 ? "2" : undefined}
            >
              {old && /\d/.test(old) && (
                <span className="t-digit-old" aria-hidden="true">{old}</span>
              )}
              <span className="t-digit">{ch}</span>
            </span>
          );
        })}
      </span>
    </>
  );
}
